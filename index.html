        // 倒數計時器邏輯 Target: 2026-12-06 12:30:00

        const targetDate = new Date('2026-12-06T12:30:00').getTime();


        function updateCountdown() {

            const now = new Date().getTime();

            const distance = targetDate - now;


            if (distance < 0) {

                document.getElementById('days').innerText = "00";

                document.getElementById('hours').innerText = "00";

                document.getElementById('minutes').innerText = "00";

                document.getElementById('seconds').innerText = "00";

                return;

            }


            const days = Math.floor(distance / (1000 * 60 * 60 * 24));

            const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

            const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));

            const seconds = Math.floor((distance % (1000 * 60)) / 1000);


            document.getElementById('days').innerText = String(days).padStart(2, '0');

            document.getElementById('hours').innerText = String(hours).padStart(2, '0');

            document.getElementById('minutes').innerText = String(minutes).padStart(2, '0');

            document.getElementById('seconds').innerText = String(seconds).padStart(2, '0');

        }


        // 初始化倒數

        setInterval(updateCountdown, 1000);

        updateCountdown();


        // 交通指引 Accordion 折疊邏輯

        function toggleAccordion(id) {

            const content = document.getElementById(id);

            const icon = document.getElementById('icon-' + id);

            

            if (!content || !icon) return;


            if (content.classList.contains('active')) {

                content.classList.remove('active');

                icon.style.transform = 'rotate(0deg)';

            } else {

                // 收合其他項目

                document.querySelectorAll('.accordion-content').forEach(item => item.classList.remove('active'));

                document.querySelectorAll('[id^="icon-acc"]').forEach(ic => ic.style.transform = 'rotate(0deg)');

                

                content.classList.add('active');

                icon.style.transform = 'rotate(180deg)';

            }

        }

        // 使用非同步 fetch (no-cors) 直接背景傳送資料至 Google 表單，不跳轉頁面

        async function handleFormSubmit(e) {

            e.preventDefault(); // 阻止原生導頁或跳轉

            

            const form = e.target;

            const submitBtn = document.getElementById('submitBtn');

            

            if (submitBtn) {

                submitBtn.disabled = true;

                submitBtn.innerHTML = '<i class="fa-solid fa-spinner fa-spin mr-2"></i>傳送中...';

            }


            try {

                // 將表單資料轉換為 URLSearchParams 格式 (支援多選核取方塊)

                const formData = new FormData(form);

                const searchParams = new URLSearchParams();

                for (const [key, value] of formData.entries()) {

                    searchParams.append(key, value);

                }


                // 直接帶入 URLSearchParams 物件，瀏覽器會自動帶入正確的 application/x-www-form-urlencoded Content-Type

                await fetch(form.action, {

                    method: 'POST',

                    mode: 'no-cors',

                    body: searchParams

                });


                // 成功後彈出提示視窗

                showSuccessModal();

            } catch (error) {

                console.error('Submit error:', error);

                // 即使發生網路異狀也彈出提示

                showSuccessModal();

            }

        }
